function getLang() {
  const lang = (navigator.language || 'en').slice(0, 2).toLowerCase();
  return content[lang] ? lang : 'en';
}

function pickRandom(arr) {
  return arr[Math.floor(Math.random() * arr.length)];
}

function init() {
  const lang = getLang();
  const c = content[lang];

  document.documentElement.lang = lang;
  document.getElementById('eyebrow').textContent = c.eyebrow;
  document.getElementById('message').innerHTML = c.message;
  document.getElementById('donate-btn').textContent = c.donate;
  document.getElementById('footer').textContent = c.footer;

  const quoteEl = document.getElementById('quote');
  quoteEl.textContent = pickRandom(c.quotes);

  setInterval(() => {
    quoteEl.classList.add('fading');
    setTimeout(() => {
      quoteEl.textContent = pickRandom(c.quotes);
      quoteEl.classList.remove('fading');
    }, 500);
  }, 8000);
}

init();
